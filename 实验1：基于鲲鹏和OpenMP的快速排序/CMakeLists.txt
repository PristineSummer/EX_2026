cmake_minimum_required(VERSION 3.16)
project(kunpeng_openmp_quicksort LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenMP 4.5 REQUIRED COMPONENTS CXX)
add_library(sort_core STATIC code/sort_core.cpp)
target_include_directories(sort_core PUBLIC code)
target_link_libraries(sort_core PUBLIC OpenMP::OpenMP_CXX)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    # Some MinGW FindOpenMP versions detect the compiler flag but omit the
    # import library. Let the GCC driver select its matching runtime at link.
    target_link_options(sort_core PUBLIC -fopenmp)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(sort_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(quicksort code/quick_sort.cpp)
target_link_libraries(quicksort PRIVATE sort_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(sort_tests tests/sort_tests.cpp)
    target_link_libraries(sort_tests PRIVATE sort_core)
    add_test(NAME sort_correctness COMMAND sort_tests)
    set_tests_properties(sort_correctness PROPERTIES TIMEOUT 180)
endif()
